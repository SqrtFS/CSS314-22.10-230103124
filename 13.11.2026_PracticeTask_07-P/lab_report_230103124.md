### harness.py

```python
from dataclasses import dataclass
import multiprocessing as mp
import time
from typing import Callable


@dataclass
class BenchStats:
    min_ms: float
    median_ms: float
    iqr_ms: float


def fnv1a_hash(s: str) -> int:
    h = 14695981039346656037
    for b in s.encode("utf-8"):
        h = ((h ^ b) * 1099511628211) & 0xFFFFFFFFFFFFFFFF
    return h


def stats_from_samples(samples) -> BenchStats:
    s = sorted(samples)
    n = len(s)
    return BenchStats(s[0], s[n // 2], s[(3 * n) // 4] - s[n // 4])


def run_benchmark(fn: Callable[[], None], warmup: int = 2, runs: int = 7) -> BenchStats:
    for _ in range(warmup):
        fn()
    samples = []
    for _ in range(runs):
        t0 = time.perf_counter_ns()
        fn()
        samples.append((time.perf_counter_ns() - t0) / 1e6)
    samples.sort()
    med = samples[runs // 2]
    return BenchStats(samples[0], med, samples[(3 * runs) // 4] - samples[runs // 4])


def emit_verify_tag(phase: str, seed: int, m1: float, m2: float) -> str:
    hw = mp.cpu_count()
    sig = fnv1a_hash(f"{phase}|{seed}|{hw}|{m1:.4f}|{m2:.4f}")
    return f"VERIFY_HASH[{phase}]: {sig:016X} (P_log={hw})"


def p_series(plog: int):
    ps, p = [], 1
    while p <= plog:
        ps.append(p)
        p *= 2
    if plog not in ps:
        ps.append(plog)
    return ps


def save_md(fname: str, lines) -> None:
    with open(fname, "w", encoding="utf-8") as f:
        f.write("\n".join(lines) + "\n")
```

```python
import ctypes
import multiprocessing as mp
import time
from multiprocessing import shared_memory

from config import STUDENT_ID, P_PHYS, QUICK, CACHE_LINE
from harness import stats_from_samples, emit_verify_tag, p_series, save_md

M = 1_600_000 if QUICK else 16_000_000
WARMUP, RUNS = (0, 3) if QUICK else (1, 5)
STRIDES = [1, 2, 4, 8, 16, 32]


def fs_worker(shm_name, slot_idx, iters, n_slots, barrier):
    shm = shared_memory.SharedMemory(name=shm_name)
    arr = (ctypes.c_uint64 * n_slots).from_buffer(shm.buf)
    acc = arr[slot_idx]
    barrier.wait(120)
    for i in range(iters):
        acc = (acc * 6364136223846793005 + (i | 1)) & 0xFFFFFFFFFFFFFFFF
        arr[slot_idx] = acc
    del arr
    shm.close()


def run_once(p, stride):
    n_slots = max(512, p * stride)
    shm = shared_memory.SharedMemory(create=True, size=n_slots * 8)
    try:
        barrier = mp.Barrier(p + 1)
        iters = M // p
        procs = [
            mp.Process(target=fs_worker,
                       args=(shm.name, t * stride, iters, n_slots, barrier))
            for t in range(p)
        ]
        for pr in procs:
            pr.start()
        barrier.wait(120)
        t0 = time.perf_counter_ns()
        for pr in procs:
            pr.join()
        return (time.perf_counter_ns() - t0) / 1e6
    finally:
        shm.close()
        shm.unlink()


def measure(p, stride):
    for _ in range(WARMUP):
        run_once(p, stride)
    return stats_from_samples([run_once(p, stride) for _ in range(RUNS)])


def karp_flatt(sp, p):
    return (1.0 / sp - 1.0 / p) / (1.0 - 1.0 / p)


def main():
    plog = mp.cpu_count()
    out = [f"## Phase 1 - false sharing (M={M}, P_phys={P_PHYS}, P_log={plog})", ""]

    out += [f"### Stride sweep, p = {P_PHYS}", "",
            "| stride | bytes | median ms | IQR ms | Mops/s |",
            "|---|---|---|---|---|"]
    sweep = {}
    for st in STRIDES:
        s = measure(P_PHYS, st)
        sweep[st] = s
        mops = M / (s.median_ms * 1e3)
        out.append(f"| {st} | {st * 8} | {s.median_ms:.1f} | {s.iqr_ms:.1f} | {mops:.2f} |")
        print(out[-1], flush=True)
    drops = [(STRIDES[i], STRIDES[i + 1], sweep[STRIDES[i]].median_ms / sweep[STRIDES[i + 1]].median_ms)
             for i in range(len(STRIDES) - 1)]
    a, b, r = max(drops, key=lambda x: x[2])
    out += ["", f"Largest drop in time: stride {a} ({a * 8} B) -> stride {b} ({b * 8} B), x{r:.2f} faster. "
                f"Cache line = {CACHE_LINE} B (stride {CACHE_LINE // 8}).", ""]
    print(out[-2], flush=True)

    ps = p_series(plog)
    t1 = {}
    for st in (1, 16):
        out += [f"### Process scaling, stride = {st}", "",
                "| p | median ms | IQR ms | S_p | E_p | Karp-Flatt e(p) |",
                "|---|---|---|---|---|---|"]
        for p in ps:
            s = measure(p, st)
            if p == 1:
                t1[st] = s.median_ms
            sp = t1[st] / s.median_ms
            ep = sp / p
            e = f"{karp_flatt(sp, p):.4f}" if p >= 2 else "-"
            out.append(f"| {p} | {s.median_ms:.1f} | {s.iqr_ms:.1f} | {sp:.2f} | {ep:.2f} | {e} |")
            print(out[-1], flush=True)
        out.append("")

    tag = emit_verify_tag("PHASE1", STUDENT_ID, sweep[1].median_ms, sweep[16].median_ms)
    out += [tag]
    print(tag)
    save_md("results_task1.md", out)


if __name__ == "__main__":
    main()
```

### task2_scaling.py

```python
import math
import multiprocessing as mp

from config import STUDENT_ID, P_PHYS, P_BOUNDARY, QUICK
from harness import run_benchmark, emit_verify_tag, save_md

N_STRONG = 120_000 if QUICK else 1_200_000
N0 = 20_000 if QUICK else 200_000
WARMUP, RUNS = (0, 3) if QUICK else (1, 5)


def stage_a(n):
    acc = 0.0
    for i in range(int(0.05 * n)):
        x = i * 1e-6 + 0.1
        for _ in range(5):
            x = math.sqrt(math.sin(x) + 2.0)
        acc += x
    return acc


def stage_b_chunk(se):
    s, e = se
    acc = 0.0
    for i in range(s, e):
        x = i * 1e-6 + 0.1
        for _ in range(10):
            x = x * 1.0000001 + 0.5
            x = math.sin(x) + math.cos(x)
        acc += x
    return acc


def pool_warm(_):
    return 0


def trial(pool, n, p):
    stage_a(n)
    chunks = [(n * t // p, n * (t + 1) // p) for t in range(p)]
    return sum(pool.map(stage_b_chunk, chunks, chunksize=1))


def bench_trial(n, p):
    with mp.Pool(processes=p) as pool:
        pool.map(pool_warm, range(2 * p))
        return run_benchmark(lambda: trial(pool, n, p), WARMUP, RUNS).median_ms


def inflection(S, plog, name, out):
    b = P_BOUNDARY
    if b >= plog:
        out.append(f"{name}: boundary {b} >= P_log, no inflection.")
        return
    before = S[b] - S.get(b - 1, 0.0)
    after = S[b + 1] - S[b]
    out.append(f"{name}: dS({b - 1}->{b}) = {before:.3f}, dS({b}->{b + 1}) = {after:.3f}, "
               f"marginal gain changes by {100 * (after / before - 1):+.1f}% across the P-core/E-core boundary (p={b}->{b + 1}).")
    print(out[-1], flush=True)


def main():
    plog = mp.cpu_count()
    out = [f"## Phase 2 - scaling (P_phys={P_PHYS}, P_log={plog})", ""]

    ta1 = run_benchmark(lambda: stage_a(N_STRONG), WARMUP, RUNS).median_ms
    T, S = {}, {}
    out += [f"### Strong scaling, N = {N_STRONG}", "",
            "| p | T ms | S_p | E_p | dS vs p-1 |", "|---|---|---|---|---|"]
    for p in range(1, plog + 1):
        T[p] = bench_trial(N_STRONG, p)
        S[p] = T[1] / T[p]
        d = f"{S[p] - S[p - 1]:.3f}" if p > 1 else "-"
        out.append(f"| {p} | {T[p]:.1f} | {S[p]:.2f} | {S[p] / p:.2f} | {d} |")
        print(out[-1], flush=True)
    s_frac = ta1 / T[1]
    out += ["", f"T_StageA(1) = {ta1:.1f} ms, s = {s_frac:.4f}, Amdahl cap S_inf = 1/s = {1 / s_frac:.1f}; "
                f"observed S_{plog} = {S[plog]:.2f}", ""]
    print(out[-2], flush=True)
    inflection(S, plog, "Strong", out)
    out.append("")

    Tw, Sg = {}, {}
    out += [f"### Weak scaling, N0 = {N0}", "",
            "| p | N(p) | T ms | E_weak | T_StageA(p) ms | alpha | S_Gustafson |", "|---|---|---|---|---|---|---|"]
    for p in range(1, plog + 1):
        n = p * N0
        Tw[p] = bench_trial(n, p)
        tap = run_benchmark(lambda: stage_a(n), WARMUP, RUNS).median_ms
        alpha = tap / Tw[p]
        Sg[p] = p - alpha * (p - 1)
        out.append(f"| {p} | {n} | {Tw[p]:.1f} | {Tw[1] / Tw[p]:.2f} | {tap:.1f} | {alpha:.4f} | {Sg[p]:.2f} |")
        print(out[-1], flush=True)
    out.append("")
    inflection({p: Tw[1] / Tw[p] * p for p in Tw}, plog, "Weak (scaled throughput)", out)
    out.append("")

    tag = emit_verify_tag("PHASE2", STUDENT_ID, S[plog], Sg[plog])
    out.append(tag)
    print(tag)
    save_md("results_task2.md", out)


if __name__ == "__main__":
    main()
```

```python
import multiprocessing as mp
import random
from multiprocessing import shared_memory

from config import STUDENT_ID, P_PHYS, QUICK, L1D_BYTES, L2_BYTES
from harness import run_benchmark, emit_verify_tag, save_md

BUF = 8 * 10 ** 6 if QUICK else 64 * 10 ** 6

WARM_BW, RUNS_BW = (1, 5)

BS_LIST = [8, 16, 32, 64, 96, 128]

_mv_a = _mv_b = _keep = None


def init_copy(na, nb):
    global _mv_a, _mv_b, _keep
    sa = shared_memory.SharedMemory(name=na)
    sb = shared_memory.SharedMemory(name=nb)
    _keep = (sa, sb)
    _mv_a, _mv_b = sa.buf, sb.buf


def copy_worker(se):
    s, e = se
    _mv_a[s:e] = _mv_b[s:e]
    return 0


def fma_worker(iters):
    b, c = 1.0000001, 1e-9
    a0 = a1 = a2 = a3 = a4 = a5 = a6 = a7 = 1.0
    a8 = a9 = a10 = a11 = a12 = a13 = a14 = a15 = 1.1
    for _ in range(iters):
        a0 = a0 * b + c
        a1 = a1 * b + c
        a2 = a2 * b + c
        a3 = a3 * b + c
        a4 = a4 * b + c
        a5 = a5 * b + c
        a6 = a6 * b + c
        a7 = a7 * b + c
        a8 = a8 * b + c
        a9 = a9 * b + c
        a10 = a10 * b + c
        a11 = a11 * b + c
        a12 = a12 * b + c
        a13 = a13 * b + c
        a14 = a14 * b + c
        a15 = a15 * b + c
    return a0 + a1 + a2 + a3 + a4 + a5 + a6 + a7 + a8 + a9 + a10 + a11 + a12 + a13 + a14 + a15


_A = _B = _C = _shms = None


def init_gemm(na, nb, nc, n):
    global _A, _B, _C, _shms
    _shms = [shared_memory.SharedMemory(name=x) for x in (na, nb, nc)]
    _A, _B, _C = [s.buf[: n * n * 8].cast("d") for s in _shms]


def gemm_worker(task):
    v, bs, r0, r1, N = task
    A, B, C = _A, _B, _C
    if v == 1:
        for i in range(r0, r1):
            ia = i * N
            for j in range(N):
                s = 0.0
                for k in range(N):
                    s += A[ia + k] * B[k * N + j]
                C[ia + j] = s
    elif v == 2:
        for i in range(r0, r1):
            ia = i * N
            for j in range(N):
                C[ia + j] = 0.0
            for k in range(N):
                a_ik = A[ia + k]
                kb = k * N
                for j in range(N):
                    C[ia + j] += a_ik * B[kb + j]
    else:
        for i in range(r0, r1):
            for j in range(N):
                C[i * N + j] = 0.0
        for ii in range(r0, r1, bs):
            imax = min(ii + bs, r1)
            for jj in range(0, N, bs):
                jmax = min(jj + bs, N)
                for kk in range(0, N, bs):
                    kmax = min(kk + bs, N)
                    for i in range(ii, imax):
                        ia = i * N
                        for k in range(kk, kmax):
                            a_ik = A[ia + k]
                            kb = k * N
                            for j in range(jj, jmax):
                                C[ia + j] += a_ik * B[kb + j]
    return 0


def main():
    p = P_PHYS
    out = [f"## Phase 3 - Roofline and GEMM (p = P_phys = {p}, buffers {BUF / 1e6:.0f} MB x2, GEMM N={GN})", ""]

    sa = shared_memory.SharedMemory(create=True, size=BUF)
    sb = shared_memory.SharedMemory(create=True, size=BUF)
    sb.buf[:BUF] = b"\x01" * BUF
    sa.buf[:BUF] = b"\x00" * BUF
    try:
        ranges = [(BUF * t // p, BUF * (t + 1) // p) for t in range(p)]
        with mp.Pool(p, initializer=init_copy, initargs=(sa.name, sb.name)) as pool:
            st = run_benchmark(lambda: pool.map(copy_worker, ranges, chunksize=1), WARM_BW, RUNS_BW)
    finally:
        sa.close()
        sb.close()
        sa.unlink()
        sb.unlink()
    bw = 2 * BUF / ((st.median_ms / 1e3) * 1e9)
    out += ["### 3A - sustained DRAM bandwidth",
            f"- median {st.median_ms:.2f} ms (IQR {st.iqr_ms:.2f}) -> BW_sustained = {bw:.2f} GB/s", ""]
    print(out[-2], flush=True)

    iters = int(FMA_FLOPS / 32)
    with mp.Pool(p) as pool:
        st = run_benchmark(lambda: pool.map(fma_worker, [iters] * p, chunksize=1), 1, 3)
    psus = p * iters * 32 / ((st.median_ms / 1e3) * 1e9)
    ridge = psus / bw
    out += ["### 3A - compute ceiling",
            f"- median {st.median_ms:.1f} ms -> P_sustained = {psus:.4f} GFLOP/s (aggregate over {p} procs)",
            f"- Ridge point I* = P/BW = {ridge:.5f} FLOP/byte", ""]
    print(out[-3], flush=True)

    N = GN
    nbytes = N * N * 8
    shm = [shared_memory.SharedMemory(create=True, size=nbytes) for _ in range(3)]
    memA, memB, memC = [s.buf[:nbytes].cast("d") for s in shm]
    rng = random.Random(STUDENT_ID)
    for i in range(N * N):
        memA[i] = rng.uniform(-1, 1)
        memB[i] = rng.uniform(-1, 1)
    rows = [(N * t // p, N * (t + 1) // p) for t in range(p)]

    def run_variant(pool, v, bs):
        return run_benchmark(
            lambda: pool.map(gemm_worker, [(v, bs, r0, r1, N) for r0, r1 in rows], chunksize=1),
            WARM_G, RUNS_G)

    def gflops(ms):
        return 2 * N ** 3 / (ms * 1e6)

    res = {}
    try:
        with mp.Pool(p, initializer=init_gemm, initargs=(shm[0].name, shm[1].name, shm[2].name, N)) as pool:
            st1 = run_variant(pool, 1, 0)
            C1 = memC.tolist()
            st2 = run_variant(pool, 2, 0)
            C2 = memC.tolist()
            res["v1"], res["v2"] = st1, st2
            d12 = max(abs(x - y) for x, y in zip(C1, C2))
            out += ["### 3B - variants (N=%d)" % N, "",
                    "| variant | median ms | IQR ms | GFLOP/s | I_eff = GFLOP/s / BW |", "|---|---|---|---|---|",
                    f"| 1 naive i-j-k | {st1.median_ms:.1f} | {st1.iqr_ms:.1f} | {gflops(st1.median_ms):.4f} | {gflops(st1.median_ms) / bw:.5f} |",
                    f"| 2 i-k-j | {st2.median_ms:.1f} | {st2.iqr_ms:.1f} | {gflops(st2.median_ms):.4f} | {gflops(st2.median_ms) / bw:.5f} |",
                    "", f"max|C1-C2| = {d12:.3e}", "",
                    "### 3B - tile sweep (Variant 3)", "",
                    "| BS | W_tile KiB | fits | median ms | IQR ms | GFLOP/s | max|C3-C2| | I_tiled=BS/12 | roof=min(P,I*BW) |",
                    "|---|---|---|---|---|---|---|---|---|"]
            print("\n".join(out[-9:]), flush=True)
            sweep = {}
            for bs in BS_LIST:
                st3 = run_variant(pool, 3, bs)
                d = max(abs(x - y) for x, y in zip(memC.tolist(), C2))
                w = 3 * bs * bs * 8
                fit = "L1d" if w <= L1D_BYTES else ("L2" if w <= L2_BYTES else ">L2")
                it = bs / 12
                roof = min(psus, it * bw)
                ok = "OK" if d < 1e-9 else "FAIL"
                sweep[bs] = st3
                out.append(f"| {bs} | {w / 1024:.1f} | {fit} | {st3.median_ms:.1f} | {st3.iqr_ms:.1f} | "
                           f"{gflops(st3.median_ms):.4f} | {d:.1e} {ok} | {it:.2f} | {roof:.4f} |")
                print(out[-1], flush=True)
    finally:
        for m in (memA, memB, memC):
            m.release()
        for s in shm:
            try:
                s.close()
            except BufferError:
                pass
            s.unlink()

    best = min(sweep, key=lambda b: sweep[b].median_ms)
    g1, g3 = gflops(res["v1"].median_ms), gflops(sweep[best].median_ms)
    out += ["", f"BS* = {best} (W_tile = {3 * best * best * 8 / 1024:.1f} KiB; "
                f"L1d = {L1D_BYTES // 1024} KiB, L2 = {L2_BYTES // 1024} KiB)",
            f"Variant 1 = {g1:.4f} GFLOP/s, Variant 3 (BS*) = {g3:.4f} GFLOP/s, speedup x{g3 / g1:.2f}", ""]
    print(out[-3], flush=True)
    print(out[-2], flush=True)
```

### task4_scheduling.py

```python
import math
import multiprocessing as mp
import time

from config import STUDENT_ID, P_PHYS, QUICK
from harness import emit_verify_tag, save_md

N = 800 if QUICK else 8000
RUNS = 3
CHUNKS = [1, 8, 64, 512]


def work_item(i):
    k = 1 + (i * i) // 1600
    x = 0.001 * i + 0.5
    acc = 0.0
    for _ in range(k):
        x = math.sin(x) + math.cos(x)
        acc += math.sqrt(abs(x) + 1.0)
    return acc


def static_worker(t, p, mode, tw, barrier):
    idx = range(N * t // p, N * (t + 1) // p) if mode == "block" else range(t, N, p)
    barrier.wait(300)
    t0 = time.perf_counter()
    acc = 0.0
    for i in idx:
        acc += work_item(i)
    tw[t] = time.perf_counter() - t0


def dynamic_worker(t, chunk, counter, tw, barrier):
    barrier.wait(300)
    t0 = time.perf_counter()
    acc = 0.0
    while True:
        with counter.get_lock():
            start = counter.value
            counter.value = start + chunk
        if start >= N:
            break
        for i in range(start, min(start + chunk, N)):
            acc += work_item(i)
    tw[t] = time.perf_counter() - t0


def run_config(kind, p, chunk=None):
    tw = mp.Array("d", p)
    barrier = mp.Barrier(p + 1)
    if kind == "dynamic":
        counter = mp.Value("i", 0)
        procs = [mp.Process(target=dynamic_worker, args=(t, chunk, counter, tw, barrier)) for t in range(p)]
    else:
        procs = [mp.Process(target=static_worker, args=(t, p, kind, tw, barrier)) for t in range(p)]
    for pr in procs:
        pr.start()
    barrier.wait(300)
    t0 = time.perf_counter()
    for pr in procs:
        pr.join()
    wall = time.perf_counter() - t0
    times = list(tw)
    lam = max(times) / (sum(times) / p)
    return wall * 1e3, lam, times


def median_run(kind, p, chunk=None):
    runs = sorted((run_config(kind, p, chunk) for _ in range(RUNS)), key=lambda r: r[0])
    return runs[RUNS // 2]


def main():
    p = P_PHYS
    total_items = sum(1 + (i * i) // 1600 for i in range(N))
    out = [f"## Phase 4 - scheduling (N={N}, p={p}, total heavy iterations = {total_items})", "",
           "| strategy | T_wall ms | lambda | min worker s | max worker s |", "|---|---|---|---|---|"]
    results = {}
    configs = [("block", None, "Block static"), ("cyclic", None, "Cyclic static")] + \
              [("dynamic", c, f"Dynamic C={c}") for c in CHUNKS]
    for kind, c, label in configs:
        wall, lam, times = median_run(kind, p, c)
        results[label] = (wall, lam)
        out.append(f"| {label} | {wall:.1f} | {lam:.3f} | {min(times):.3f} | {max(times):.3f} |")
        print(out[-1], flush=True)
    best_label = min((l for l in results if l.startswith("Dynamic")), key=lambda l: results[l][0])
    out += ["", f"C* = {best_label.split('=')[1]} (lowest T_wall among dynamic)", ""]
    print(out[-2], flush=True)
    tag = emit_verify_tag("PHASE4", STUDENT_ID, results["Block static"][0], results[best_label][0])
    out.append(tag)
    print(tag)
    save_md("results_task4.md", out)


if __name__ == "__main__":
    main()
```
